Plugging the real agents in

This page renders whatever is pushed to window.MachineWire. The demo button replays a scripted run through the same API, so replacing it with live events is one line: subscribe to your agent's event stream and forward each event.

// from the buyer agent or the seller endpoint (any transport — WebSocket, SSE, polling)
new WebSocket("ws://localhost:8787/events").onmessage = e => MachineWire.push(JSON.parse(e.data));

// event shape
{
  lane:  "buyer" | "match" | "wire" | "sellers" | "ledger",
  kind:  "reason" | "request" | "response" | "payment" | "tx" | "refusal" | "done",
  title: "Paying 0.5 XRP",           // one line
  detail:"Within per-call cap 1 XRP", // optional second line
  code:  402 | 200 | 422,             // wire responses only
  body:  { ... },                     // wire: rendered as JSON
  tx: { type:"VaultDeposit", amount:"60 XRP", hash:"...", status:"pending"|"validated", setup:false },
  // match lane: { category:"lend_borrow", band:"moderate", candidates:[{id, name, rating, netYield, capacity, score, out?:"reason"}] }
  // sellers lane: { id, status:"listed"|"matched"|"filtered", reason? }
  // wire lane: add  from:"buyer"|"platform"|"seller", to:"buyer"|"platform"|"seller"
}
MachineWire.updateTx(hash, "validated");  // flip a pending tx when it validates
MachineWire.reset();
